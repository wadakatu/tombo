# tombo.js Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `tombo.js` — an optional, zero-dependency state layer providing 3-state theme persistence and auto-injected code-block copy buttons — plus its CSS, migrated examples, and docs, under a progressive-enhancement invariant.

**Architecture:** One classic IIFE script (`tombo.js`, non-module, loaded with `defer`) does all post-load wiring via one delegated `document` click listener; a ~100-byte inline `<head>` snippet applies the stored theme pre-paint to avoid a flash. Theme controls are author markup that the script reveals + wires; copy buttons are injected by the script (so they only exist when JS runs). A minimal `window.Tombo = { setTheme, getTheme }` is the sanctioned programmatic hook.

**Tech Stack:** Vanilla ES5-compatible JavaScript (no build, no framework), CSS custom properties, `localStorage`, `navigator.clipboard`. Verification is browser-based (chrome-devtools MCP) + `scripts/check_contrast.py`.

**Spec:** `docs/superpowers/specs/2026-07-23-tombo-js-design.md`

## Global Constraints

Every task implicitly includes these (verbatim from the spec / repo conventions):

- **Zero dependency, no build step.** Do NOT add a JS test runner, bundler, or `package.json`. Adding jsdom/vitest violates the constraint. Tests are browser behavioral checks via chrome-devtools + the contrast gate.
- **Distribution form:** classic script, **not** an ES module; loaded with `defer`; IIFE, no global pollution except `window.Tombo`.
- **PE invariant (must hold at every step):** with no JS — theme follows OS via `light-dark()`, the `.tombo-themes` control stays `hidden`, no copy button exists, the page is fully usable.
- **Adoption gate:** Baseline *widely available* only. (`localStorage`, `navigator.clipboard`, `dataset`, `closest`, event delegation all qualify.)
- **No color/token changes.** `scripts/check_contrast.py` must still exit 0. No `#000`/`#FFF`, no shadows with offset (spec §5).
- **Theme storage:** key `tombo-theme`; values `"tombo"` | `"hotaru"` only; `system` = key absent (`removeItem`).
- **Commits:** Conventional Commits in English; **no** `Co-Authored-By`; end every commit message with the trailer `Claude-Session: https://claude.ai/code/session_01CYKP1wE7CusnhuQm9CxTue`. Branch `feature/tombo-js`. Verify visual changes in BOTH themes (`data-theme` = `tombo` and `hotaru`).

- [ ] **Step 2: Write tombo.js (theme half)**

Create `tombo.js` with exactly this content (copy injection is added in Task 2; leave the `injectCopy`/`copyFrom`/`liveRegion`/`announce` calls out until then — the delegated listener's copy branch and `injectCopy()` in `init` are added in Task 2):

```js
/*! TOMBO tombo.js v0.6.0 — wadakatu design system. Optional, zero-dependency. MIT. https://github.com/wadakatu/tombo */
(function () {
  "use strict";
  var KEY = "tombo-theme";
  var VALID = { system: 1, tombo: 1, hotaru: 1 };

  function stored() {
    try { var t = localStorage.getItem(KEY); return (t === "tombo" || t === "hotaru") ? t : "system"; }
    catch (e) { return "system"; }
  }
  function apply(setting) {
    var root = document.documentElement;
    if (setting === "tombo" || setting === "hotaru") root.dataset.theme = setting;
    else delete root.dataset.theme;
  }
  function reflect() {
    var cur = stored(), btns = document.querySelectorAll("[data-tombo-theme]");
    for (var i = 0; i < btns.length; i++)
      btns[i].setAttribute("aria-pressed", String(btns[i].getAttribute("data-tombo-theme") === cur));
  }
  function setTheme(setting) {
    if (!VALID[setting]) return;
    try { if (setting === "system") localStorage.removeItem(KEY); else localStorage.setItem(KEY, setting); }
    catch (e) {}
    apply(setting); reflect();
  }
  function getTheme() { return stored(); }

  document.addEventListener("click", function (e) {
    var t = e.target.closest && e.target.closest("[data-tombo-theme]");
    if (t) { setTheme(t.getAttribute("data-tombo-theme")); }
  });

  function init() {
    apply(stored());
    var c = document.querySelectorAll(".tombo-themes[hidden]");
    for (var i = 0; i < c.length; i++) c[i].hidden = false;
    reflect();
  }
  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", init);
  else init();

  window.Tombo = { setTheme: setTheme, getTheme: getTheme };
})();
```

- [ ] **Step 2: Add the copy code to tombo.js**

Add these four functions inside the IIFE, immediately after `getTheme` (before the `document.addEventListener("click", …)` line):

```js
  var live;
  function announce(msg) {
    if (!live) {
      live = document.createElement("div");
      live.className = "tombo-sr-only";
      live.setAttribute("role", "status");
      live.setAttribute("aria-live", "polite");
      document.body.appendChild(live);
    }
    live.textContent = ""; live.textContent = msg;
  }
  function injectCopy() {
    var blocks = document.querySelectorAll(".tombo-code:not([data-tombo-copy='off'])");
    for (var i = 0; i < blocks.length; i++) {
      var fig = blocks[i];
      if (fig.querySelector(".tombo-copy")) continue;
      var cap = fig.querySelector("figcaption");
      if (!cap) { cap = document.createElement("figcaption"); fig.insertBefore(cap, fig.firstChild); }
      var btn = document.createElement("button");
      btn.type = "button"; btn.className = "tombo-copy";
      btn.setAttribute("aria-label", "Copy code"); btn.textContent = "COPY ⧉";
      cap.appendChild(btn);
    }
  }
  function copyFrom(btn) {
    var fig = btn.closest(".tombo-code"); if (!fig) return;
    var pre = fig.querySelector("pre"); if (!pre) return;
    var done = function () {
      btn.textContent = "COPIED ✓"; announce("Copied");
      setTimeout(function () { btn.textContent = "COPY ⧉"; }, 1500);
    };
    var fail = function () { announce("Press Ctrl or Cmd + C to copy"); };
    try {
      if (navigator.clipboard && navigator.clipboard.writeText)
        navigator.clipboard.writeText(pre.textContent).then(done, fail);
      else fail();
    } catch (e) { fail(); }
  }
```

    var c = e.target.closest(".tombo-copy");
